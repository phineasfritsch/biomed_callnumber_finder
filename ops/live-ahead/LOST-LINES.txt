/* A result here answers "which edition, and which shelf". Everything else about a record --
the full subject headings, request and recall, the citation exports, a login that knows
whether you already have it out -- lives in Primo, and there is no reason to reimplement any
of it. Primo addresses a bib by its Alma MMS id with an `alma` prefix, which is the 001
already parsed out of every record: docid=alma9974748673606533. Verified against the live
pnxs API, where `recordid` comes back in exactly that form.
`src/worker.js` builds the same URL for article records, from ids that are not always MMS
numbers, so it derives `context` and encodes what it is given. The two are documented
together in ENDPOINTS.md, and Tools/catalog.test.js asserts they still agree on a bib id --
which is the only thing that stops a change to one of them from being made to one of them.
Inside the markers so it is covered: this builds a URL out of a value read off the network,
and the assertion that an 001 which is not an MMS number links nowhere should be run rather
than believed. */
const PRIMO_VIEW='https://search.library.ucla.edu/discovery/fulldisplay',
PRIMO_VID='01UCS_LAL:UCLA';
function primoHref(mms){
const id=String(mms==null?'':mms).trim();
if(!/^\d+$/.test(id)) return '';     // an 001 that is not an MMS number links nowhere
// Digits only by the line above, so there is nothing here that needs escaping.
return PRIMO_VIEW+'?docid=alma'+id+'&context=L&vid='+PRIMO_VID;
// Title, linked to the record it came from. Unlinkable records still get their title.
function recordLink(rec,cls,label){
const href=primoHref(rec.mms);
return href
? '<a class="'+cls+'" href="'+esc(href)+'" target="_blank" rel="noopener noreferrer">'+esc(label)+'</a>'
: '<span class="'+cls+'">'+esc(label)+'</span>';
h+='<div class="ti">'+recordLink(rec,'ti-l',rec.title)+'</div>';
/* A linked title is easy to miss on a card that is mostly call numbers, so the way out to
Primo is named in the chip row too, where the reader is already looking. */
if(href) tags+='<a class="catlink" href="'+esc(href)+'" target="_blank" rel="noopener noreferrer">'+
'In the catalog</a>';
